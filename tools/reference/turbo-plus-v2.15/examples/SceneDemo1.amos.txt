Screen Open 0,320,192,32,Lowres
Screen Display 0,135,50,320,192
Flash Off : Curs Off : Hide On : Cls 0
Load "SpaceIcons.Abk"
Get Icon Palette 
 Scene Load Dir$+"Space2.Scene",6
 Scene Convert 6 To 5
 Scene Bank 5
 Scene 16 View 0,0,0 To 320,192
Pen 1 : Paper 0
_PRINT
Do 
   A$=Inkey$
   If A$="1"
      SDRAWING
   End If 
   If A$="2"
      SSCROLLING
   End If 
   If A$="3"
      CREDITS
   End If 
   If A$="4"
      Erase 2
      Erase 6
      Cls 0
      Locate 0,12 : Centre "Bye!"
      Wait 100
      Erase 2
      Erase 6
      End 
   End If 
Loop

Procedure SSCROLLING
Cls 0
Locate 0,0 : Centre "Speed Scrolling"
Locate 0,4 : Centre "320x192x32 screen / 16x16 icons"
Locate 0,14 : Print "I will show you the difference between"
Locate 0,15 : Print "Amos and Turbo Plus scene commands."
Locate 0,20 : Centre "Press any key or mouse..."
Repeat 
Until Inkey$<>"" or Mouse Key<>0
MAPX=0
T=Timer
For MAPX=0 To 180
   For Y=0 To 11
      For X=0 To 19
         Paste Icon X*16,Y*16,Peek(Start(6)+4+(X+MAPX)+Y*200)+1
      Next X
   Next Y
   If Left Click Then Exit 
Next MAPX
TIME=Timer-T
Cls 0
Locate 0,0 : Centre "Regular Amos and Amos Pro"
Locate 0,4 : Centre "Boy, was that S L O W !!!"
'no-because you can exit early -Locate 0,6 : Print "It took ";TIME/60;" seconds!"  
Locate 0,18 : Centre "Press anything for a REAL speed demon!"
Wait 50
Repeat 
Until Inkey$<>"" or Mouse Key<>0
Cls 0
T=Timer
'we use scene do to draw the entire scene which is really unnecessary. 
'its a bit slower than using scene Left/Right, etc, but much easier and
'no extra BASIC code no matter which direction you scroll, so maybe even faster this way!
For MAPX=0 To 180
    Scene 16 Do MAPX,0
Next 
TIME=Timer-T
Cls 0
Locate 0,0 : Centre "Turbo Plus Scene commands"
Locate 0,4 : Centre "Now that's a lot better!"
Locate 0,6 : Print "It only took ";TIME/60;" seconds!"
Locate 0,18 : Centre "Press anything..."
Repeat 
Until Inkey$<>"" or Mouse Key<>0
_PRINT
End Proc
